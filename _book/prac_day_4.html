4.2.2 2. A rewrite of code using dplyr and the pipe

flights1 <- flights |> filter(month == 1)
carrier_vec <- flights1 |> distinct(carrier) |> pull(carrier)
carrier_dist_vec_mean <- numeric(length(carrier_vec))
carrier_dist_vec_sd <- numeric(length(carrier_vec))

for (i in seq_along(carrier_vec)) {
  carrier_data <- flights1 |> filter(carrier == carrier_vec[i])
  carrier_dist_vec_mean[i] <- carrier_data |> summarize(mean_dist = mean(distance)) |> pull(mean_dist)
  carrier_dist_vec_sd[i] <- carrier_data |> summarize(sd_dist = sd(distance)) |> pull(sd_dist)
}
  dist_tbl <- tibble(
  carrier = carrier_vec,
  mean_distance = carrier_dist_vec_mean,
  sd_distance = carrier_dist_vec_sd
)
  dist_tbl |> arrange(mean_distance)

4.2.3 3. Explanation of the NA and 0 standard deviations

It could be the case that the “OO” carrier had only 1 observation, making the standard deviation calculation undefined, hence NA. The other carrier, ‘YV’, might have all the distances equal.
